defmodule Loopex.LLM.ReqLLM.InProcess.PackagedDriver do
  @moduledoc """
  ## Concept

  Runs embedded calls and ask admission from code extracted from the shipped
  escript.

  ## Technical depth

  This module is compiled into the fixture directory, not the escript. The
  plain OTP child starts with no Elixir installation or checkout code path.
  Its only non-OTP paths are the escript's extracted ebin directories and this
  one driver directory. All credentials and endpoints are synthetic.
  """

  defmodule Policy do
    @moduledoc false
    @behaviour Loopex.Policy

    @impl true
    def decide(_request), do: {:allow, nil}
  end

  @project_apps [
    :loopex,
    :loopex_app_server,
    :loopex_cli,
    :loopex_composition,
    :loopex_daemon,
    :loopex_executor_local,
    :loopex_llm_reqllm,
    :loopex_protocol,
    :loopex_reference_client,
    :loopex_store_local,
    :loopex_telemetry
  ]

  @durable_apps [
    :loopex,
    :loopex_cli,
    :loopex_composition,
    :loopex_daemon,
    :loopex_executor_local,
    :loopex_llm_reqllm,
    :loopex_protocol,
    :loopex_store_local
  ]

  def main do
    [mode, model, base, ca, workspace, extracted] =
      :init.get_plain_arguments() |> Enum.map(&List.to_string/1)

    extracted = Path.expand(extracted)
    :code.del_path(~c".")
    assert_packaged_paths(extracted)

    unless :persistent_term.get(:llm_db_store, nil) == nil,
      do: raise("catalog populated before packaged startup")

    started = fn -> Application.started_applications() |> Enum.map(&elem(&1, 0)) end
    unless Enum.all?([:req_llm, :req, :finch], &(&1 not in started.())),
      do: raise("provider dependency auto-started")

    {:ok, _} = Application.ensure_all_started(:elixir)

    if mode in ["ask_invalid", "ask_durable"] do
      run_ask_startup(mode, workspace, started)
    else
      run_provider(mode, model, base, ca, workspace, extracted, started)
    end
  end

  defp run_provider(mode, model, base, ca, workspace, extracted, started) do
    {:ok, _} = Application.ensure_all_started(:public_key)

    :ok = :public_key.cacerts_load(String.to_charlist(ca))

    initial =
      case mode do
        "skip" ->
          Application.put_env(:llm_db, :skip_packaged_load, true)
          nil

        "empty" ->
          {:ok, _} = LLMDB.load_empty()
          :persistent_term.get(:llm_db_store)

        "preloaded" ->
          {:ok, _} =
            LLMDB.load(
              custom: %{
                packaging_fixture: [
                  name: "Host fixture",
                  models: %{"host-model" => %{capabilities: %{chat: true}}}
                ]
              }
            )

          :persistent_term.get(:llm_db_store)

        "normal" ->
          nil

        "ask_ephemeral" ->
          nil
      end

    unless System.get_env("LOOPEX_PACKAGED_DOTENV_CANARY") == nil,
      do: raise("ambient dotenv canary present")

    options = [
      policy: Policy,
      model: model,
      base_url: base,
      cwd: workspace,
      tools: :none,
      max_tokens: 32,
      deadline_ms: 15_000,
      timeout: 20_000
    ]

    count = if String.starts_with?(model, "anthropic:") and mode == "normal", do: 2, else: 1

    catalogs =
      for _ <- 1..count do
        if mode == "ask_ephemeral" do
          run_ephemeral_ask(model, base, started)
        else
          case apply(:"Elixir.LoopexComposition.Ephemeral", :run, [
                 "Answer with fixture reply.",
                 options
               ]) do
            {:ok, %{outcome: :completed, text: "packaged answer"}} ->
              :ok

            other ->
              raise "packaged run failed: #{inspect(other)}"
          end
        end

        :persistent_term.get(:llm_db_store, nil)
      end

    unless Enum.all?([:req_llm, :req, :finch], &(&1 in started.())),
      do: raise("guarded provider startup did not complete")

    unless :logger in started.(), do: raise("packaged Logger did not start")

    unless Application.get_env(:req_llm, :load_dotenv) == false,
      do: raise("guarded provider startup left dotenv enabled")

    unless System.get_env("LOOPEX_PACKAGED_DOTENV_CANARY") == nil,
      do: raise("packaged startup loaded dotenv")

    assert_packaged_paths(extracted)

    after_call = List.last(catalogs)

    case {mode, String.starts_with?(model, "anthropic:")} do
      {"normal", true} ->
        unless is_map(after_call) and map_size(after_call.snapshot.models) > 0,
          do: raise("Anthropic did not load embedded catalog")

        unless hd(catalogs) == after_call,
          do: raise("Anthropic changed catalog epoch or content between calls")

        {:ok, expected} = LLMDB.Loader.load()

        unless same_catalog?(after_call.snapshot, expected),
          do: raise("Anthropic changed embedded catalog content")

      {"normal", false} ->
        unless after_call == nil, do: raise("non-Anthropic call loaded catalog")
        {:ok, _} = LLMDB.load()
        loaded = :persistent_term.get(:llm_db_store)
        {:ok, expected} = LLMDB.Loader.load()

        unless is_map(loaded) and map_size(loaded.snapshot.models) > 0 and
                 same_catalog?(loaded.snapshot, expected) do
          raise "positive embedded-catalog control failed"
        end

      {"skip", _} ->
        unless after_call == nil, do: raise("skip-packaged-load control was ignored")

      {"ask_ephemeral", _} ->
        unless after_call == nil, do: raise("ephemeral ask loaded unrelated catalog")

      {mode, _} when mode in ["empty", "preloaded"] ->
        unless after_call == initial, do: raise("host catalog was replaced")
    end

    IO.puts("M6_PACKAGED_#{String.upcase(mode)}_OK")
  end

  defp run_ephemeral_ask(model, base, started) do
    assert_project_apps([], started)

    result =
      run_packaged_ask(
        [
          "ask",
          "--policy",
          "allow-all",
          "--tools",
          "read-only",
          "--model",
          model,
          "Answer with fixture reply."
        ],
        install_interrupt: fn _, _ -> {:ok, self()} end,
        signal_manager: fn -> self() end,
        handler_live: fn _, _ -> true end,
        interrupt_phase: fn _, _ -> :idle end,
        finish_interrupt: fn _, _ -> {:ok, :ordinary} end,
        start_session: fn options ->
          assert_project_apps([], started)
          apply(LoopexComposition.Ephemeral, :start_session, [
            Keyword.put(options, :base_url, base)
          ])
        end
      )

    unless result == %{status: 0, stdout: "packaged answer\n", stderr: "ending completed\n"},
      do: raise("packaged ephemeral ask failed: #{inspect(result)}")

    unless :loopex_composition in started.() and :loopex_cli not in started.(),
      do: raise("ephemeral ask started the wrong application graph")
  end

  defp run_ask_startup(mode, workspace, started) do
    assert_project_apps([], started)

    case mode do
      "ask_invalid" ->
        for command <- ["ask", "-p"] do
          result = run_packaged_ask([command, "--output", "wrong", "fixture"])

          unless result.status == 1 and result.stdout == "" and
                   result.stderr == "loopex: invalid_output\n" do
            raise "packaged invalid ask returned wrong diagnostic: #{inspect(result)}"
          end

          assert_project_apps([], started)
        end

      "ask_durable" ->
        result =
          run_packaged_ask(
            ["-p", "--policy", "allow-all", "--state-root", Path.join(workspace, "root"), "fixture"],
            durable_run: fn _, _, _ ->
              assert_project_apps(@durable_apps, started)
              assert_load_only_dependencies(started)
              %{status: 0, stdout: "durable fixture\n", stderr: ""}
            end
          )

        unless result == %{status: 0, stdout: "durable fixture\n", stderr: ""},
          do: raise("packaged durable ask failed: #{inspect(result)}")

        assert_project_apps(@durable_apps, started)
        assert_load_only_dependencies(started)
    end

    IO.puts("M6_PACKAGED_#{String.upcase(mode)}_OK")
  end

  # Concept: this driver resolves host APIs from the extracted escript graph.
  # Technical depth: its compiler VM intentionally has no CLI/composition code
  # path. Invoke the exact public API at runtime rather than statically checking
  # that deliberately absent graph or suppressing undefined-function warnings.
  defp run_packaged_ask(arguments, options \\ []),
    do: apply(LoopexCli.Ask, :run, [arguments, options])

  defp assert_project_apps(expected, started) do
    actual = Enum.filter(@project_apps, &(&1 in started.()))
    unless actual == expected, do: raise("wrong project application graph: #{inspect(actual)}")
  end

  defp assert_load_only_dependencies(started) do
    unless Enum.all?([:req_llm, :req, :finch], &(&1 not in started.())),
      do: raise("durable startup auto-started provider dependencies")
  end

  defp same_catalog?(actual, expected) do
    Map.delete(actual, :meta) == Map.delete(expected, :meta) and
      Map.delete(actual.meta, :loaded_at) == Map.delete(expected.meta, :loaded_at)
  end

  defp assert_packaged_paths(extracted) do
    otp = :code.root_dir() |> List.to_string() |> Path.expand()

    for path <- :code.get_path() do
      path = path |> List.to_string() |> Path.expand()

      unless path == otp or String.starts_with?(path, otp <> "/") or
               String.starts_with?(path, extracted <> "/") do
        raise "foreign code path in packaged VM"
      end
    end
  end
end
